Table 예약 {
  예약_번호 uuid [pk]
  차량_번호 uuid [not null, ref: > 차량.차량_번호]
  게스트_번호 uuid [not null, ref: > 사용자.사용자_번호]
  예약당시_호스트_번호 uuid [not null, ref: > 사용자.사용자_번호]
  대여_시작_일시 timestamptz [not null]
  대여_종료_일시 timestamptz [not null]
  예약_상태 varchar(20) [not null, note: '요청, 확정, 취소, 완료 등']
  예약당시_일일요금 bigint [not null]
  차량_대여료 bigint [not null]
  보험가입_여부 boolean [not null, default: false]
  보험사명 varchar(100)
  보험상품명 varchar(100)
  자기부담금_선택 varchar(50)
  보험료 bigint [not null, default: 0]
  보험가입_상태 varchar(20)
  보장_시작_일시 timestamptz
  보장_종료_일시 timestamptz
  할인액 bigint [not null, default: 0]
  추가요금 bigint [not null, default: 0]
  총_예약_금액 bigint [not null]
  플랫폼_수수료_예정액 bigint [not null]
  호스트_예상_정산액 bigint [not null]
  예약_일시 timestamptz [not null]
  취소_일시 timestamptz

  indexes {
    (차량_번호, 대여_시작_일시, 대여_종료_일시)
    게스트_번호
    예약당시_호스트_번호
  }

  checks {
    `대여_종료_일시 > 대여_시작_일시` [name: '예약기간_유효성']
    `게스트_번호 <> 예약당시_호스트_번호` [name: '호스트게스트_중복방지']
    `차량_대여료 >= 0 AND 보험료 >= 0 AND 할인액 >= 0 AND 추가요금 >= 0 AND 총_예약_금액 >= 0` [name: '예약금액_음수방지']
    `플랫폼_수수료_예정액 >= 0 AND 호스트_예상_정산액 >= 0` [name: '수수료정산액_음수방지']
  }
}
